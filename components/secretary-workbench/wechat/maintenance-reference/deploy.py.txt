"""Explicit new-module-only transfer to existing AUP SSH host; no default action."""
import pathlib,sys,json,subprocess,base64,hashlib
def main():
 if sys.argv[1:]!=['--execute']:print('Usage: python3 -m operations.deploy --execute. Transfers only operations/*.py.');return
 root=pathlib.Path(__file__).parent
 files=[{'name':p.name,'data':base64.b64encode(p.read_bytes()).decode(),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in root.glob('*.py')]
 code="""import pathlib,json,sys,base64,hashlib,os
os.umask(0o077);root=pathlib.Path('/var/lib/secretary-wechat/operations-20261027/operations');root.mkdir(parents=True,exist_ok=True,mode=0o700)
for x in json.load(sys.stdin):
 assert pathlib.Path(x['name']).name==x['name'] and x['name'].endswith('.py');data=base64.b64decode(x['data']);assert hashlib.sha256(data).hexdigest()==x['sha256'];p=root/x['name'];assert not p.is_symlink();p.write_bytes(data);p.chmod(0o600)
print('operations source transferred and checksum checked')
"""
 p=subprocess.run(['ssh','-oBatchMode=yes','-oConnectTimeout=10','secretary-wechat-host',"python3 -c '"+code.replace("'","'\\''")+"'"],input=json.dumps(files),text=True,capture_output=True,timeout=35)
 if p.returncode:raise RuntimeError('explicit source deployment failed')
 print(p.stdout.strip())
if __name__=='__main__':main()
