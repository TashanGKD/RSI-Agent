"""Install/stop only the independent explicit-request executor; no source or model timers."""
import argparse,os,subprocess
from pathlib import Path
NAME='wechat-operations.service'
CODE='/var/lib/secretary-wechat/operations-20261027'
STATE='/var/lib/secretary-wechat/operations-state'
def unit():
 return '''[Unit]
Description=Explicit queued official WeChat operations
StartLimitIntervalSec=60
StartLimitBurst=5

[Service]
Type=simple
WorkingDirectory='''+CODE+'''
ExecStart=/usr/bin/python3 -m operations.cli run --execute --root '''+STATE+''' --config '''+STATE+'''/config.json --seconds 3600
Restart=always
RestartSec=5
UMask=0077
NoNewPrivileges=yes
TimeoutStopSec=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=default.target
'''
def main():
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('action',nargs='?',choices=['install','stop','status','print']);p.add_argument('--execute',action='store_true');a=p.parse_args()
 if not a.action or (a.action!='print' and not a.execute):p.print_help();return
 if a.action=='print':print(unit());return
 if a.action=='install':
  if str(Path(__file__).resolve().parents[1])!=CODE:raise ValueError('install only in the fixed isolated AUP deployment')
  q=Path.home()/'.config/systemd/user';q.mkdir(mode=0o700,parents=True,exist_ok=True)
  if any(x.is_symlink() for x in [q,*q.parents]):raise ValueError('service directory symlink')
  path=q/NAME
  if path.exists() and path.read_text()!=unit():raise ValueError('existing differing service is not overwritten')
  path.write_text(unit());path.chmod(0o600)
  subprocess.run(['systemctl','--user','daemon-reload'],check=True)
  subprocess.run(['systemctl','--user','enable','--now',NAME],check=True)
 elif a.action=='stop':subprocess.run(['systemctl','--user','disable','--now',NAME],check=True)
 subprocess.run(['systemctl','--user','show',NAME,'--property=ActiveState,SubState,MainPID,NRestarts'],check=True)
if __name__=='__main__':main()
